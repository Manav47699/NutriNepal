# pulling, creaitng side branch, pushing, merging, then delteing the side branch


# 1. Stash all local changes (including untracked files) so nothing gets lost
git stash -u

# 2. Switch to main and pull latest remote code safely
git checkout main
git pull origin main

# 3. Create and switch to your new side branch
git checkout -b feature/my-feature

# 4. Bring back your local files/changes
git stash pop

# 5. Stage, commit, and push the side branch
git add .
git commit -m "feat: my changes"
git push -u origin feature/my-feature

# 6. Switch back to main, merge side branch, and push to GitHub
git checkout main
git merge feature/my-feature
git push origin main

# 7. Delete the local side branch
git branch -d feature/my-feature
